  curl -fsSL https://alps-project.pages.dev/install.sh | sh

remove_begin
  if [ -n "$TERMUX_VERSION" ] || [ "$PREFIX" = "/data/data/com.termux/files/usr" ]; then
    rm -f "$PREFIX/bin/alps"
    rm -f "$PREFIX/share/fish/vendor_completions.d/alps.fish"
    rm -f "$PREFIX/etc/bash_completion.d/alps"
    rm -f "$PREFIX/share/zsh/site-functions/_alps"
  else
    sudo rm -f /usr/local/bin/alps
    sudo rm -f sudo rm -f /usr/share/bash-completion/completions/alps
    sudo rm -f /usr/share/fish/vendor_completions.d/alps.fish
    sudo rm -f /usr/share/zsh/site-functions/_alps
  fi

  if [ -n "$TERMUX_VERSION" ] || [ "$PREFIX" = "/data/data/com.termux/files/usr" ]; then
    rm -rf "$PREFIX/etc/alps"
    rm -rf "$HOME/.config/alps"
    rm -rf "$PREFIX/var/cache/alps"
  else
    sudo rm -rf /etc/alps
    rm -rf "$HOME/.config/alps"
    sudo rm -rf /var/cache/alps
  fi
