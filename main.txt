os         = linux, termux

  curl -fsSL https://alps-project.pages.dev/install.sh | sh
cmd_end
upgrade_begin
  curl -fsSL https://alps-project.pages.dev/install.sh | sh
upgrade_end
remove_begin
  if [ -n "$TERMUX_VERSION" ] || [ -n "$PREFIX" ]; then rm -f "$PREFIX/bin/alps" "$PREFIX/share/fish/vendor_completions.d/alps.fish" "$PREFIX/etc/bash_completion.d/alps" "$PREFIX/share/zsh/site-functions/_alps"; else sudo rm -f /usr/local/bin/alps /usr/share/bash-completion/completions/alps /usr/share/fish/vendor_completions.d/alps.fish /usr/share/zsh/site-functions/_alps; fi
remove_end
purge_begin
  if [ -n "$TERMUX_VERSION" ] || [ -n "$PREFIX" ]; then rm -rf "$PREFIX/etc/alps" "$HOME/.config/alps" "$PREFIX/var/cache/alps"; else sudo rm -rf /etc/alps "$HOME/.config/alps" /var/cache/alps; fi
purge_end
